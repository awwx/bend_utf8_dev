MAIN RESULTS, ALL PROVED UNCONDITIONALLY
  Det.reduce_det_fragment_proved : reduce_det_fragment, i.e.
    forall hs s f es hs1 s1 f1 es1 hs2 s2 f2 es2,
      cfg_ok s es ->
      reduce hs s f es hs1 s1 f1 es1 -> reduce hs s f es hs2 s2 f2 es2 ->
      (hs1, s1, f1, es1) = (hs2, s2, f2, es2).
  Main.fragment_deterministic_outcome :
    forall hs s f es hs' s' f' vs, cfg_ok s es ->
      reduce_trans (hs,s,f,es) (hs',s',f',v_to_e_list vs) ->
      deterministic_outcome (hs,s,f,es) (hs',s',f',v_to_e_list vs)
  where deterministic_outcome c D :=
      (forall c', reduce_trans c c' -> reduce_trans c' D)            (* no trap, no other result *)
   /\ (forall run : nat -> cfg, run 0 = c ->
         (forall n, reduce_tuple (run n) (run n.+1)) -> False)        (* no infinite run *)
   /\ (forall c', reduce_trans c c' -> (forall c'', ~ reduce_tuple c' c'') -> c' = D).
  ReduceDet.still_reaches_stuck_is_target (no premises, any configurations):
    (forall c', reduce_trans c c' -> reduce_trans c' D) ->
    forall c', reduce_trans c c' -> (forall c'', ~ reduce_tuple c' c'') -> c' = D
    (the first conjunct of deterministic_outcome implies the third).
  The decoder's instance (decoder_call_outcome, for the call configurations
  of ../decoder/Boundary.v's call_returns) is ../decoder/DetInstance.v.
